\section{Research Problems in Function Theory — Problem 6.100}\label{statement}

\subsection{Definitions and mathematical statement}

\paragraph{Shared notation.}$\mathbb N=\{1,2,\ldots\}$, and $\mathbb Z,\mathbb Q,\mathbb R,\mathbb C$ denote the integers, rationals, reals and complex numbers. Any other conventions are specified locally.


Write $\mathbb D=\{z\in\mathbb C:|z|<1\}$ and $\mathbb T=\partial\mathbb D$. Let $S$ be the class of holomorphic injective functions $f:\mathbb D\to\mathbb C$ with $f(0)=0$ and $f'(0)=1$. For a nonconstant holomorphic function $f$ on a domain $D$, its valence is $v_D(f)=\sup_{w\in\mathbb C}n_D(w,f)\in\mathbb N\cup\{\infty\}$, where $n_D(w,f)$ counts the zeros of $f-w$ in $D$ with multiplicity. The function is $p$-valent if $v_D(f)\leq p$. For $0<\alpha<1$ put $\beta=1-\alpha$ and, for $f,g\in S$, define
\[F_\alpha(z)=\alpha f(z)+\beta g(z),\qquad
G_\alpha(z)=z\exp\!\left(\alpha\log\frac{f(z)}z+\beta\log\frac{g(z)}z\right).\]
The quotients extend to value one at zero and never vanish. Each logarithm is the holomorphic branch with value zero at zero. Put $a_*=(1+e^{\pi})^{-1}$.
\paragraph{Statement recorded in the supplied source.}
For $0<\alpha\leq a_*$, do finite bounds $B_F(\alpha)$ and $B_G(\alpha)$ exist such that
\[v_{\mathbb D}(F_\alpha)\leq B_F(\alpha),\qquad v_{\mathbb D}(G_\alpha)\leq B_G(\alpha)
\quad\text{for every } f,g\in S?\]
Determine the valence behavior on both fringes $(0,a_*]$ and $[1-a_*,1)$; interchanging $f$ and $g$ interchanges $\alpha$ and $1-\alpha$. \sref{statement}{2}
\subsection{Short English statement}
Determine whether arithmetic and geometric means of normalized univalent functions have uniformly bounded valence for weights near either endpoint.

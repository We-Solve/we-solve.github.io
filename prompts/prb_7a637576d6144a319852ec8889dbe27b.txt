<<<
\documentclass{article}
\usepackage{amsmath,amssymb,hyperref}
\begin{document}
\section*{Cohen--Lenstra--Gerth rank moments of quadratic class groups}
\subsection{Definitions and mathematical statement}
For a fundamental quadratic discriminant $D$, let
$C_D=\operatorname{Cl}^+(\mathbb Q(\sqrt D))$. This narrow ideal class group
is the group of fractional ideals modulo principal ideals generated by
elements positive in every real embedding. Put $C_D^2=\{a^2:a\in C_D\}$.
For a finite abelian group $B$ define
$r_\ell(B)=\dim_{\mathbb F_\ell}(B/B^\ell)$.

Let $\mathcal D^-(X)$ and $\mathcal D^+(X)$ be the negative and positive
fundamental quadratic discriminants with absolute value at most $X$.
For each fixed prime $\ell$ and integer $k\geq1$, set
\[
 F_{\ell,k}(D)=\prod_{i=0}^{k-1}
     \bigl(\ell^{r_\ell(C_D^2)}-\ell^i\bigr).
\]
Do both limits exist and satisfy
\[
 \lim_{X\to\infty}\frac{\sum_{D\in\mathcal D^-(X)}F_{\ell,k}(D)}
 {|\mathcal D^-(X)|}=1,\qquad
 \lim_{X\to\infty}\frac{\sum_{D\in\mathcal D^+(X)}F_{\ell,k}(D)}
 {|\mathcal D^+(X)|}=\ell^{-k}?
\]
The case $\ell=2$ is known for every $k$, as is $(\ell,k)=(3,1)$;
the displayed question retains the full prime and moment range.

\subsection{Short English statement}
Do the stated rank moments of squared narrow quadratic class groups have
the predicted averages, separately for real and imaginary quadratic fields?

\subsection{Sources}
J. Kl\"uners, contribution on Cohen--Lenstra heuristics to
\emph{Arithmetic and Differential Galois Groups}, Oberwolfach Report
26/2007, pp.\ 1476--1478, Conjecture 1 and Theorem 3,
\url{https://ems.press/content/serial-article-files/46110}.
The extension to $\ell=2$ is due to F. Gerth; the proved $2$-primary moment
case is due to E. Fouvry and J. Kl\"uners.
For recent context see J. Ellenberg, \emph{Recent Progress around
Cohen--Lenstra Heuristics}, arXiv:2606.06024,
\url{https://arxiv.org/abs/2606.06024}.
This is an independently written mathematical formulation.
\end{document}

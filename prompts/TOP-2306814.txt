\section{Research Problems in Function Theory — Problem 6.14$\prime$}\label{2306814}
{\small\sffamily UnsolvedMath ID 2306814 · Analysis}
\subsection{Definitions and mathematical statement}

\paragraph{Shared notation.}$\mathbb N=\{1,2,\ldots\}$, and $\mathbb Z,\mathbb Q,\mathbb R,\mathbb C$ denote the integers, rationals, reals and complex numbers. Any other conventions are specified locally.


Write $\mathbb D=\{z\in\mathbb C:|z|<1\}$ and $\mathbb T=\partial\mathbb D$. For $p>0$ and a holomorphic $f$ on $\mathbb D$, let $n_f(w)$ count the preimages of $w$ with multiplicity. The function is areally mean $p$-valent when
\[\int_0^{2\pi}\int_0^R n_f(\rho e^{i\theta})\rho\,d\rho\,d\theta\leq\pi pR^2\qquad(R>0).\]
Writing $f(z)=\sum_{j=0}^{\infty}a_jz^j$, put $\mu_p(f)=\max\{|a_j|:j\in\mathbb Z,\ 0\leq j\leq p\}$, and let $S(p)$ be the mean $p$-valent functions with $\mu_p(f)=1$.
For a mean $p$-valent function $f(z)=\sum_{j\geq0}a_jz^j$ and an integer $k\geq1$, form its coefficient Hankel determinant
\[A_n^{(k)}(f)=\det(a_{n+i+j})_{0\leq i,j<k}.\]
An exponent $a$ is admissible for $(p,k)$ if $|A_n^{(k)}(f)|^{1/k}=O_f(n^a)$ as $n\to\infty$ for every mean $p$-valent $f$; its bound can depend on $f,p,k$. Let $j_k(p)$ be the infimum of admissible exponents.
\paragraph{Statement recorded in the supplied source.}
Determine the best possible exponents $j_k(p)$ for the mean $p$-valent class and whether the corresponding boundary bounds hold. The source extends equation(6.6) to this class with the admissible exponent
\[-\frac12+16\sqrt{\frac{p^3}{k}},\]
and asks for the sharp improvement. The normalization $\mu_p=1$ is not required for the individual-function asymptotic estimate. \sref{2306814}{2}
\subsection{Short English statement}
Find the sharp power growth of coefficient Hankel determinants for mean-valent functions.

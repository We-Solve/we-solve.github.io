\section{For each k and every sufficiently large n with k dividing ((n-1) choose (k-1)), there is a universal cycle for the k-sub}\label{statement}

\subsection{Definitions and mathematical statement}

\paragraph{Shared notation.}$\mathbb N=\{1,2,\ldots\}$, and $\mathbb Z,\mathbb Q,\mathbb R,\mathbb C$ denote the integers, rationals, reals and complex numbers. Any other conventions are specified locally.


For integers $n\ge k\ge1$, write $[n]=\{1,\ldots,n\}$ and $\binom{[n]}k$ for its family of $k$-element subsets. A universal cycle for this family is a cyclic word $a_0a_1\cdots a_{L-1}$ over $[n]$, of length $L=\binom nk$, such that every cyclic window
\[\{a_j,a_{j+1},\ldots,a_{j+k-1}\}\qquad(0\le j<L),\]
with subscripts reduced modulo $L$, has $k$ distinct symbols and these $L$ sets are exactly the members of $\binom{[n]}k$, each once.

\paragraph{Statement recorded in the supplied source.}
For every integer $k\ge1$, does there exist an integer $n_0(k)\ge k$ such that for every $n\ge n_0(k)$ satisfying $k\mid\binom{n-1}{k-1}$, a universal cycle for $\binom{[n]}k$ exists? \sref{statement}{2}

\subsection{Short English statement}
For each fixed subset size and all sufficiently large ground sets, is the necessary divisibility condition sufficient for a cyclic word whose windows represent every subset exactly once?

\item[\hypertarget{source:statement:2}{S2}] Yevgeniy Rudoy, \emph{An inductive approach to constructing universal cycles on the $k$-subsets of $[n]$} (2013 revision), Section1, definition, Lemma1.1 and Chung--Diaconis--Graham conjecture, pages1--2; Section2, cycles and universal cycles, pages3--4. \url{https://arxiv.org/abs/1209.4662}.
